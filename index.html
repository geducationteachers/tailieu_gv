/** * G.EDU – KHO TÀI LIỆU GV (Backend) * ------------------------------------------------------------ * Chức năng: * - API cho trang GEDU_TaiLieu_GV.html (đăng nhập GV / Admin, * danh mục tài liệu, thêm/sửa/xoá, báo lỗi, xử lý báo lỗi) * - Trình chiếu file HTML trên Drive (mở là chạy, không cần tải về) * * CÀI ĐẶT: * 1. Tạo 1 Google Sheet mới → Tiện ích mở rộng → Apps Script → dán file này. * 2. Sửa ADMIN_PASS bên dưới. * 3. Chọn hàm setup → Chạy (cấp quyền) → Sheet tự tạo 2 tab TAILIEU, FEEDBACK. * 4. Triển khai → Tùy chọn triển khai mới → Ứng dụng web * Thực thi với tư cách: Tôi | Người có quyền truy cập: Bất kỳ ai * 5. Copy link /exec dán vào API_URL trong file HTML. * (Sửa code sau này: Triển khai → Quản lý → ✎ → Phiên bản mới, giữ nguyên link) */ const ADMIN_PASS = 'admin@gedu'; // mật khẩu riêng của admin (GV không cần đăng nhập) const SHEET_ID = ''; // để '' nếu script tạo từ trong Sheet const TZ = 'Asia/Ho_Chi_Minh'; const SH_FILES = 'TAILIEU'; const SH_FB = 'FEEDBACK'; const FILE_HEAD = ['ID', 'Sách', 'Unit', 'Tiêu đề unit', 'Tên file', 'Loại', 'Link', 'Thứ tự', 'Cập nhật']; const FB_HEAD = ['ID', 'Thời gian', 'Giáo viên', 'File ID', 'Sách', 'Unit', 'Tên file', 'Loại lỗi', 'Nội dung', 'Trạng thái', 'Ghi chú admin', 'Cập nhật']; /* ================= SETUP ================= */ function setup() { sh_(SH_FILES, FILE_HEAD); sh_(SH_FB, FB_HEAD); } function ss_() { return SHEET_ID ? SpreadsheetApp.openById(SHEET_ID) : SpreadsheetApp.getActiveSpreadsheet(); } function sh_(name, head) { const ss = ss_(); let s = ss.getSheetByName(name); if (!s) { s = ss.insertSheet(name); s.appendRow(head); s.setFrozenRows(1); s.getRange(1, 1, 1, head.length).setFontWeight('bold').setBackground('#eeeeee'); if (name === SH_FILES) s.getRange('C:C').setNumberFormat('@'); // Unit giữ dạng chữ } return s; } /* ================= ĐỌC / GHI SHEET ================= */ function read_(name) { const s = sh_(name, name === SH_FILES ? FILE_HEAD : FB_HEAD); const v = s.getDataRange().getValues(); const h = v.shift(); return v.map((r, i) => { const o = { _row: i + 2 }; h.forEach((k, j) => { o[k] = r[j] instanceof Date ? Utilities.formatDate(r[j], TZ, 'dd/MM/yyyy HH:mm') : r[j]; }); return o; }).filter(o => String(o['ID']).trim() !== ''); } function writeRow_(s, head, row, obj) { s.getRange(row, 1, 1, head.length).setValues([head.map(h => obj[h])]); } function fileObj_(r) { return { id: String(r['ID']), book: String(r['Sách']), unit: String(r['Unit']), title: String(r['Tiêu đề unit'] || ''), name: String(r['Tên file']), type: String(r['Loại']).toLowerCase() === 'html' ? 'html' : 'pdf', link: String(r['Link'] || ''), order: Number(r['Thứ tự']) || 0, updated: r['Cập nhật'] }; } function repObj_(r) { return { id: String(r['ID']), time: r['Thời gian'], teacher: r['Giáo viên'], fileId: String(r['File ID']), book: r['Sách'], unit: String(r['Unit']), fileName: r['Tên file'], issue: r['Loại lỗi'], content: r['Nội dung'], status: r['Trạng thái'] || 'Mới', note: r['Ghi chú admin'] || '', updated: r['Cập nhật'] }; } function driveId_(u) { u = String(u || ''); const m = u.match(/\/d\/([\w-]{20,})/) || u.match(/[?&]id=([\w-]{20,})/); return m ? m[1] : ''; } /* ================= TRÌNH CHIẾU HTML ================= */ function doGet(e) { const p = (e && e.parameter) || {}; if (p.id) return view_(p.id); if (p.u) return viewUrl_(p.u); return msg_('G.EDU – API Kho tài liệu đang hoạt động.'); } function view_(id) { const inList = read_(SH_FILES).some(r => driveId_(r['Link']) === id); if (!inList) return msg_('File không có trong danh mục tài liệu.'); let html = '', title = 'G.EDU'; try { const f = DriveApp.getFileById(id); html = f.getBlob().getDataAsString('UTF-8'); title = f.getName(); } catch (err) { // File không nằm trong Drive của tài khoản chạy script → thử tải bản công khai const res = UrlFetchApp.fetch('https://drive.google.com/uc?export=download&id=' + id, { muteHttpExceptions: true, followRedirects: true }); if (res.getResponseCode() !== 200) { return msg_('Không mở được file. Hãy chia sẻ file ở chế độ "Bất kỳ ai có đường liên kết".'); } html = res.getContentText('UTF-8'); } return render_(html, title); } /* Link tải file từ nơi khác (không phải Google Drive) */ function viewUrl_(u) { u = String(u).trim(); const inList = read_(SH_FILES).some(r => String(r['Link']).trim() === u); if (!inList) return msg_('File không có trong danh mục tài liệu.'); const fetchUrl = /dropbox\.com/i.test(u) ? dropboxDl_(u, 1) : u; const res = UrlFetchApp.fetch(fetchUrl, { muteHttpExceptions: true, followRedirects: true }); if (res.getResponseCode() !== 200) return msg_('Không tải được file từ link này (mã lỗi ' + res.getResponseCode() + ').'); const name = decodeURIComponent((fetchUrl.split('?')[0].split('/').pop()) || 'G.EDU'); return render_(res.getContentText('UTF-8'), name); } /* Dropbox: dl=1 → tải nội dung file, dl=0 → trang xem trước */ function dropboxDl_(u, v) { return /[?&]dl=\d/.test(u) ? u.replace(/([?&])dl=\d/, '$1dl=' + v) : u + (u.indexOf('?') > -1 ? '&' : '?') + 'dl=' + v; } function render_(html, title) { return HtmlService.createHtmlOutput(html) .setTitle(String(title).replace(/\.html?$/i, '')) .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function msg_(t) { return HtmlService.createHtmlOutput( '
' + t + '
') .setTitle('G.EDU'); } /* ================= API ================= */ function doPost(e) { let req = {}; try { req = JSON.parse(e.postData.contents); } catch (err) { return out_({ ok: false, error: 'Dữ liệu không hợp lệ' }); } const role = req.pass === ADMIN_PASS ? 'admin' : 'gv'; try { switch (req.action) { case 'login': return role === 'admin' ? out_({ ok: true, role: role }) : out_({ ok: false, error: 'Sai mật khẩu admin' }); case 'list': return out_(list_(role)); case 'report': return out_(report_(req)); } if (role !== 'admin') return out_({ ok: false, error: 'Sai mật khẩu admin – vui lòng đăng nhập lại' }); switch (req.action) { case 'saveFile': return out_(saveFile_(req.file)); case 'deleteFile': return out_(deleteFile_(req.id)); case 'updateReport': return out_(updateReport_(req)); } return out_({ ok: false, error: 'Lệnh không hợp lệ' }); } catch (err) { return out_({ ok: false, error: String(err.message || err) }); } } function out_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function list_(role) { const files = read_(SH_FILES).map(fileObj_); const reps = read_(SH_FB).map(repObj_); const flagged = []; reps.forEach(r => { if (r.status !== 'Đã sửa' && flagged.indexOf(r.fileId) < 0) flagged.push(r.fileId); }); const res = { ok: true, role: role, files: files, flagged: flagged }; if (role === 'admin') res.reports = reps; return res; } function withLock_(fn) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { return fn(); } finally { lock.releaseLock(); } } function saveFile_(f) { f = f || {}; const need = { book: 'Sách', unit: 'Unit', name: 'Tên file', type: 'Loại' }; Object.keys(need).forEach(k => { if (!String(f[k] || '').trim()) throw new Error('Thiếu thông tin: ' + need[k]); }); return withLock_(() => { const s = sh_(SH_FILES, FILE_HEAD); const rows = read_(SH_FILES); const now = new Date(); const obj = { 'ID': f.id || ('F' + now.getTime()), 'Sách': String(f.book).trim(), 'Unit': String(f.unit).trim(), 'Tiêu đề unit': String(f.title || '').trim(), 'Tên file': String(f.name).trim(), 'Loại': f.type === 'html' ? 'html' : 'pdf', 'Link': String(f.link || '').trim(), 'Thứ tự': Number(f.order) || 0, 'Cập nhật': now }; const ex = f.id ? rows.find(r => String(r['ID']) === String(f.id)) : null; if (f.id && !ex) throw new Error('Không tìm thấy tài liệu cần sửa'); if (ex) writeRow_(s, FILE_HEAD, ex._row, obj); else s.appendRow(FILE_HEAD.map(h => obj[h])); // Đồng bộ tiêu đề unit cho các file cùng unit if (obj['Tiêu đề unit']) { const col = FILE_HEAD.indexOf('Tiêu đề unit') + 1; rows.forEach(r => { if (String(r['Sách']) === obj['Sách'] && String(r['Unit']) === obj['Unit'] && String(r['ID']) !== String(obj['ID']) && r['Tiêu đề unit'] !== obj['Tiêu đề unit']) { s.getRange(r._row, col).setValue(obj['Tiêu đề unit']); } }); } return { ok: true, id: obj['ID'] }; }); } function deleteFile_(id) { return withLock_(() => { const s = sh_(SH_FILES, FILE_HEAD); const r = read_(SH_FILES).find(x => String(x['ID']) === String(id)); if (!r) throw new Error('Không tìm thấy tài liệu'); s.deleteRow(r._row); return { ok: true }; }); } function report_(req) { const content = String(req.content || '').trim(); const teacher = String(req.teacher || '').trim(); if (!teacher) throw new Error('Thiếu tên giáo viên'); if (!content) throw new Error('Vui lòng mô tả lỗi'); const f = read_(SH_FILES).find(x => String(x['ID']) === String(req.fileId)); if (!f) throw new Error('Không tìm thấy file'); return withLock_(() => { const now = new Date(); const obj = { 'ID': 'R' + now.getTime(), 'Thời gian': now, 'Giáo viên': teacher, 'File ID': f['ID'], 'Sách': f['Sách'], 'Unit': f['Unit'], 'Tên file': f['Tên file'], 'Loại lỗi': String(req.issue || 'Khác'), 'Nội dung': content, 'Trạng thái': 'Mới', 'Ghi chú admin': '', 'Cập nhật': now }; sh_(SH_FB, FB_HEAD).appendRow(FB_HEAD.map(h => obj[h])); return { ok: true }; }); } function updateReport_(req) { return withLock_(() => { const s = sh_(SH_FB, FB_HEAD); const r = read_(SH_FB).find(x => String(x['ID']) === String(req.id)); if (!r) throw new Error('Không tìm thấy báo lỗi'); const ok = ['Mới', 'Đang xử lý', 'Đã sửa']; const st = ok.indexOf(req.status) > -1 ? req.status : 'Mới'; s.getRange(r._row, FB_HEAD.indexOf('Trạng thái') + 1).setValue(st); s.getRange(r._row, FB_HEAD.indexOf('Ghi chú admin') + 1).setValue(String(req.note || '')); s.getRange(r._row, FB_HEAD.indexOf('Cập nhật') + 1).setValue(new Date()); return { ok: true }; }); }
